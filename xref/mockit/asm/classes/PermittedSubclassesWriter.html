6   package mockit.asm.classes;
7   
8   import edu.umd.cs.findbugs.annotations.NonNull;
9   
10  import mockit.asm.constantPool.AttributeWriter;
11  import mockit.asm.constantPool.ConstantPoolGeneration;
12  import mockit.asm.util.ByteVector;
13  
14  import org.checkerframework.checker.index.qual.NonNegative;
15  
16  /**
17   * Generates the "PermittedSubclasses" attribute of a sealed class (Java 17+). The attribute is a u2 count followed by
18   * that many u2 CONSTANT_Class indices, structurally identical to "NestMembers". Preserving it verbatim lets a redefined
19   * sealed class keep the attribute the JVM requires to stay unchanged across redefinition.
20   */
21  final class PermittedSubclassesWriter extends AttributeWriter {
22      @NonNegative
23      private final int[] permittedSubclassIndices;
24  
25      PermittedSubclassesWriter(@NonNull ConstantPoolGeneration cp, @NonNull String[] permittedSubclassNames) {
26          super(cp, "PermittedSubclasses");
27  
28          permittedSubclassIndices = new int[permittedSubclassNames.length];
29  
30          for (int i = 0; i < permittedSubclassNames.length; i++) {
31              permittedSubclassIndices[i] = cp.newClass(permittedSubclassNames[i]);
32          }
33      }
34  
35      @NonNegative
36      @Override
37      public int getSize() {
38          return 8 + 2 * permittedSubclassIndices.length;
39      }
40  
41      @Override
42      public void put(@NonNull ByteVector out) {
43          int numberOfSubclasses = permittedSubclassIndices.length;
44          put(out, 2 + 2 * numberOfSubclasses);
45          out.putShort(numberOfSubclasses);
46  
47          for (int permittedSubclassIndex : permittedSubclassIndices) {
48              out.putShort(permittedSubclassIndex);
49          }
50      }
51  }
