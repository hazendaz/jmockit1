183     /**
184      * java.net.InetAddress is sealed (JDK 17+), so its classfile carries a PermittedSubclasses attribute.
185      * Round-tripping it through ClassReader -> ClassWriter must preserve that attribute; otherwise the JVM rejects a
186      * redefinition of the class with "attempted to change the class ... PermittedSubclasses attribute". This asserts
187      * the attribute survives in the emitted class-level attribute table, not merely that some bytes came out. (Checking
188      * the constant pool would be meaningless: the pool is copied wholesale, so the "PermittedSubclasses" UTF8 entry
189      * survives regardless of whether the attribute itself is written.)
190      */
191     @Test
192     void preservesPermittedSubclassesAttributeOnSealedClass() {
193         byte[] code = ClassFile.readBytesFromClassFile(java.net.InetAddress.class);
194         assertTrue(hasClassAttribute(code, "PermittedSubclasses"),
195                 "test premise: InetAddress should be sealed (carry PermittedSubclasses) on this JDK");
196 
197         ClassReader reader = new ClassReader(code);
198         ClassWriter writer = new ClassWriter(reader);
199         reader.accept(writer);
200         byte[] result = writer.toByteArray();
201 
202         assertNotNull(result);
203         assertTrue(hasClassAttribute(result, "PermittedSubclasses"),
204                 "PermittedSubclasses attribute must be preserved in the rewritten sealed class");
205     }

207     /**
208      * Structurally parses a class file down to its class-level attribute table and reports whether an attribute with
209      * the given name is present there. Unlike a constant-pool scan, this reflects what the writer actually emitted, so
210      * it distinguishes a preserved attribute from a merely-copied name entry.
211      */
212     private static boolean hasClassAttribute(byte[] cf, String attributeName) {
213         String[] utf8 = new String[readUnsignedShort(cf, 8)];
214         int index = 10;
215 
216         // Constant pool: record UTF8 values (to resolve attribute name indices) and skip everything by tag size.
217         for (int i = 1; i < utf8.length; i++) {
218             int tag = cf[index] & 0xFF;
219             index++;
220 
221             switch (tag) {
222                 case 1: // Utf8
223                     int length = readUnsignedShort(cf, index);
224                     utf8[i] = new String(cf, index + 2, length, java.nio.charset.StandardCharsets.UTF_8);
225                     index += 2 + length;
226                     break;
227                 case 5: // Long
228                 case 6: // Double
229                     index += 8;
230                     i++; // occupies two pool slots
231                     break;
232                 case 7: // Class
233                 case 8: // String
234                 case 16: // MethodType
235                 case 19: // Module
236                 case 20: // Package
237                     index += 2;
238                     break;
239                 case 15: // MethodHandle
240                     index += 3;
241                     break;
242                 default: // Integer/Float/Fieldref/Methodref/InterfaceMethodref/NameAndType/Dynamic/InvokeDynamic
243                     index += 4;
244                     break;
245             }
246         }
247 
248         index += 6; // access_flags, this_class, super_class
249         index += 2 + 2 * readUnsignedShort(cf, index); // interfaces_count + interfaces[]
250         index = skipMembers(cf, index); // fields[]
251         index = skipMembers(cf, index); // methods[]
252 
253         // Class-level attributes table: attributes_count (u2) then attribute_info[].
254         int attrCount = readUnsignedShort(cf, index);
255         index += 2;
256 
257         for (int a = 0; a < attrCount; a++) {
258             int nameIndex = readUnsignedShort(cf, index);
259             int attrLength = readInt(cf, index + 2);
260             index += 6;
261 
262             if (attributeName.equals(utf8[nameIndex])) {
263                 return true;
264             }
265 
266             index += attrLength;
267         }
268 
269         return false;
270     }
271 
272     /**
273      * Skips a fields[] or methods[] table (both have identical member_info layout), returning the offset just past it.
274      */
275     private static int skipMembers(byte[] cf, int index) {
276         int count = readUnsignedShort(cf, index);
277         index += 2;
278 
279         for (int m = 0; m < count; m++) {
280             index += 6; // access_flags, name_index, descriptor_index
281 
282             int attrCount = readUnsignedShort(cf, index);
283             index += 2;
284 
285             for (int a = 0; a < attrCount; a++) {
286                 int attrLength = readInt(cf, index + 2); // skip name_index (u2), read length (u4)
287                 index += 6 + attrLength;
288             }
289         }
290 
291         return index;
292     }
293 
294     private static int readUnsignedShort(byte[] b, int i) {
295         return (b[i] & 0xFF) << 8 | b[i + 1] & 0xFF;
296     }
297 
298     private static int readInt(byte[] b, int i) {
299         return (b[i] & 0xFF) << 24 | (b[i + 1] & 0xFF) << 16 | (b[i + 2] & 0xFF) << 8 | b[i + 3] & 0xFF;
300     }
