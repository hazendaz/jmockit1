6   package mockit.coverage;
7   
8   import static org.junit.jupiter.api.Assertions.assertEquals;
9   
10  import org.junit.jupiter.api.Test;
11  
12  final class CoveragePercentageTest {
13  
14      @Test
15      void calculateReturnsMinusOneWhenTotalIsZero() {
16          assertEquals(-1, CoveragePercentage.calculate(0, 0));
17      }
18  
19      @Test
20      void calculateReturnsZeroWhenNothingCovered() {
21          assertEquals(0, CoveragePercentage.calculate(0, 10));
22      }
23  
24      @Test
25      void calculateReturnsHundredWhenFullyCovered() {
26          assertEquals(100, CoveragePercentage.calculate(10, 10));
27      }
28  
29      @Test
30      void calculateRoundsToNearestInteger() {
31          assertEquals(33, CoveragePercentage.calculate(1, 3));
32          assertEquals(67, CoveragePercentage.calculate(2, 3));
33      }
34  
35      @Test
36      void percentageColorIsPureRedWhenNothingCovered() {
37          assertEquals("ff0000", CoveragePercentage.percentageColor(0, 10));
38      }
39  
40      @Test
41      void percentageColorIsPureGreenWhenFullyCovered() {
42          assertEquals("00ff00", CoveragePercentage.percentageColor(10, 10));
43      }
44  
45      @Test
46      void percentageColorIsIntermediateForPartialCoverage() {
47          String color = CoveragePercentage.percentageColor(5, 10);
48  
49          assertEquals(6, color.length());
50          assertEquals("00", color.substring(4));
51      }
52  }
