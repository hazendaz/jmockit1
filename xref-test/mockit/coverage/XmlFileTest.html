6   package mockit.coverage;
7   
8   import static org.junit.jupiter.api.Assertions.assertTrue;
9   
10  import java.io.File;
11  import java.io.IOException;
12  import java.nio.file.Files;
13  
14  import mockit.coverage.data.CoverageData;
15  import mockit.coverage.data.FileCoverageData;
16  
17  import org.junit.jupiter.api.Test;
18  import org.junit.jupiter.api.io.TempDir;
19  
20  final class XmlFileTest {
21      @TempDir
22      File outputDir;
23  
24      @Test
25      void generateWritesExpectedXmlContent() throws IOException {
26          CoverageData data = new CoverageData();
27          FileCoverageData fileData = data.getOrAddFile("com/example/Foo.java", "class");
28          fileData.lineCoverageInfo.addLine(5);
29          fileData.lineCoverageInfo.addLine(8);
30          fileData.lineCoverageInfo.registerExecution(5, null);
31  
32          XmlFile xmlFile = new XmlFile(outputDir.getPath(), data);
33          xmlFile.generate();
34  
35          File generatedFile = outputDir.toPath().resolve("coverage.xml").toFile();
36          assertTrue(generatedFile.exists());
37  
38          String content = Files.readString(generatedFile.toPath());
39          assertTrue(content.contains("<?xml version=\"1.0\" encoding=\"UTF-8\"?>"));
40          assertTrue(content.contains("<coverage version=\"1\">"));
41          assertTrue(content.contains("<file path=\"com/example/Foo.java\">"));
42          assertTrue(content.contains("lineNumber=\"5\" covered=\"true\""));
43          assertTrue(content.contains("lineNumber=\"8\" covered=\"false\""));
44          assertTrue(content.contains("</coverage>"));
45      }
46  
47      @Test
48      void generateWritesNoLinesForFilesWithoutExecutableLines() throws IOException {
49          CoverageData data = new CoverageData();
50          data.getOrAddFile("com/example/Empty.java", "class");
51  
52          XmlFile xmlFile = new XmlFile(outputDir.getPath(), data);
53          xmlFile.generate();
54  
55          String content = Files.readString(outputDir.toPath().resolve("coverage.xml"));
56          assertTrue(content.contains("<file path=\"com/example/Empty.java\">"));
57          assertTrue(!content.contains("lineToCover"));
58      }
59  }
