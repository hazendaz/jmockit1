296     // The following tests exercise mocking of java.net.InetAddress, which is a sealed class as of JDK 17
297     // (it permits Inet4Address and Inet6Address). Mocking a sealed class requires JMockit to redefine it
298     // while preserving its PermittedSubclasses attribute; dropping that attribute makes the JVM reject the
299     // redefinition ("attempted to change the class ... PermittedSubclasses attribute"). These end-to-end
300     // tests confirm that a sealed JRE class is fully mockable: instance-method stubbing, static-method
301     // stubbing, and recorded exceptions all behave as they did before sealed classes existed.
302 
303     /**
304      * Mocks an instance method of the sealed {@link InetAddress} class.
305      *
306      * @param mockAddress
307      *            the mock address
308      */
309     @Test
310     void mockInstanceMethodOfSealedInetAddress(@Mocked InetAddress mockAddress) {
311         new Expectations() {
312             {
313                 mockAddress.getHostName();
314                 result = "mocked-host";
315                 mockAddress.getHostAddress();
316                 result = "10.0.0.5";
317             }
318         };
319 
320         assertEquals("mocked-host", mockAddress.getHostName());
321         assertEquals("10.0.0.5", mockAddress.getHostAddress());
322     }

324     /**
325      * Mocks the static {@link InetAddress#getByName(String)} factory and stubs the resulting instance.
326      *
327      * @param anyAddress
328      *            any mocked address instance
329      *
330      * @throws Exception
331      *             the exception
332      */
333     @Test
334     void mockStaticFactoryOfSealedInetAddress(@Mocked InetAddress anyAddress) throws Exception {
335         new Expectations() {
336             {
337                 InetAddress.getByName("db.example.com");
338                 result = anyAddress;

343 
344         InetAddress resolved = InetAddress.getByName("db.example.com");

373     /**
374      * Records a checked {@link UnknownHostException} from the mocked static factory, confirming exception recording
375      * works for a sealed class.
376      *
377      * @param anyAddress
378      *            any mocked address instance
379      */
380     @Test
381     void recordCheckedExceptionFromSealedInetAddress(@Mocked InetAddress anyAddress) {

386                     result = new UnknownHostException("unresolved");
